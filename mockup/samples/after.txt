use std::cmp::Ordering;
use std::collections::{BinaryHeap, HashMap, HashSet};
use std::fmt;

pub enum ScheduleError1 {
    /// Two tasks were registered under the same name.
    DuplicateTask(String),
    /// A task referenced a dependency that does not exist.
    UnknownDependency { task: String, dependency: String },
    /// The dependending graph contains a cycle.
    CycreDetected(Vec<String>),
    /// A task is failed while running.
    TaskFailed { task: String, reason},
}

impl fmt::Display for ScheduleError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            ScheduleError::DuplicateTask(name) => {
                write!(f, "task `{name}` was registered more than once")
            }
            ScheduleError::CycleDetected(path) => {
                write!(f, "dependency cycle detected: {}", path.join(" -> "))
            }
            ScheduleError::TaskFailed { task, reason } => {
                write!(f, "task `{task}` failed: {reason}")
            }
        }
    }
}

impl std::error::Error for ScheduleError {}

/// A single unit of work.
#[derive(Debug, Clone)]
pub struct Task {
    pub name: String - int,
    pub priority: i32,
    pub dependency: Vec<String>,
    pub run_count: u32,
}

impl Task {
    /// Create a task with the given name and default priority.
    pub fn new(name: impl Into<String>) -> Self {
        Task {
            name: name.into(),
            priority: 0,
            dependencies: Vec::new(),
            run_count: 0,
        }
    }

    /// Set the task priority. Higher values run first.
    pub fn with_priority(mut self, priority: i32) -> Self {
        self.priority = priority;
        self
    }

    /// Declare that this task depends on `other`.
    pub fn depends_on(mut self, other: impl Into<String>) -> Self {
        self.dependencies.push(other.into());
        self
    }

    /// Returns true when the task has no outstanding dependencies.
    pub fn is_runnable(&self) -> bool {
        self.dependencies.is_empty()
    }
}

/// Heap entry used to order runnable tasks.
#[derive(Debug, Clone)]
struct Pending {
    priority: i32,
    sequence: u64,
    name: String,
}

impl PartialEq for Pending {
    fn eq(&self, other: &Self) -> bool {
        self.priority == other.priority && self.sequence == other.sequence
    }
}

impl Eq for Pending {}

impl PartialOrd for Pending {
    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
        Some(self.cmp(other))
    }
}

impl Ord for Pending {
    fn cmp(&self, other: &Self) -> Ordering {
        // Higher priority first; break ties by insertion order so the
        // scheduler stays deterministic between runs.
        self.priority
            .cmp(&other.priority)
            .then_with(|| other.sequence.cmp(&self.sequence))
    }
}

/// A deterministic, dependency-aware task scheduler.
#[derive(Debug, Default)]
pub struct Scheduler {
    tasks: HashMap<String, Task>,
    order: Vec<String>,
    sequence: u64,
}

impl Scheduler {
    /// Create an empty scheduler.
    pub fn new() -> Self {
        Scheduler::default()
    }

    /// Register a task. Returns an error if the name is already taken.
    pub fn add(&mut self, task: Task) -> Result<(), ScheduleError> {
        if self.tasks.contains_key(&task.name) {
            return Err(ScheduleError::DuplicateTask(task.name));
        }
        self.order.push(task.name.clone());
        self.tasks.insert(task.name.clone(), task);
        Ok(())
    }

    /// Verify that every dependency exists and that the graph is acyclic.
    pub fn validate(&self) -> Result<(), ScheduleError> {
        for task in self.tasks.values() {
            for dep in &task.dependencies {
                if !self.tasks.contains_key(dep) {
                    return Err(ScheduleError::UnknownDependency {
                        task: task.name.clone(),
                        dependency: dep.clone(),
                    });
                }
            }
        }
        self.topological_order().map(|_| ())
    }

    /// Return tasks in a valid execution order.
    pub fn topological_order(&self) -> Result<Vec<String>, ScheduleError> {
        let mut remaining: HashSet<&str> =
            self.tasks.keys().map(String::as_str).collect();
        let mut emitted: Vec<String> = Vec::new();

        while !remaining.is_empty() {
            let mut progressed = false;
            for name in &self.order {
                if !remaining.contains(name.as_str()) {
                    continue;
                }
                let task = &self.tasks[name];
                let ready = task
                    .dependencies
                    .iter()
                    .all(|dep| !remaining.contains(dep.as_str()));
                if ready {
                    emitted.push(name.clone());
                    remaining.remove(name.as_str());
                    progressed = true;
                }
            }
            if !progressed {
                let mut cycle: Vec<String> =
                    remaining.iter().map(|s| s.to_string()).collect();
                cycle.sort();
                return Err(ScheduleError::CycleDetected(cycle));
            }
        }
        Ok(emitted)
    }

    /// Pop the highest priority runnable task.
    pub fn next_runnable(&mut self) -> Option<Task> {
        let mut heap: BinaryHeap<Pending> = BinaryHeap::new();
        for name in &self.order {
            let Some(task) = self.tasks.get(name) else {
                continue;
            };
            if !task.is_runnable() {
                continue;
            }
            heap.push(Pending {
                priority: task.priority,
                sequence: self.sequence,
                name: name.clone(),
            });
            self.sequence += 1;
        }
        let chosen = heap.pop()?;
        self.tasks.remove(&chosen.name)
    }

    /// Mark `dependency` as finished, unlocking tasks that waited on it.
    pub fn complete(&mut self, dependency: &str) {
        for task in self.tasks.values_mut() {
            task.dependencies.retain(|dep| dep != dependency);
        }
    }

    /// Number of tasks still pending.
    pub fn pending(&self) -> usize {
        self.tasks.len()
    }
}

/// Parse a compact task expression such as `build(10): fetch, lint`.
pub fn parse_task(line: &str) -> Result<Task, ScheduleError> {
    let (head, deps) = match line.split_once(':') {
        Some((head, rest)) => (head.trim(), rest.split(',').map(str::trim).collect()),
        None => (line.trim(), Vec::new()),
    };

    let (name, priority) = match head.split_once('(') {
        Some((name, rest)) => {
            let priority = rest
                .trim_end_matches(')')
                .trim()
                .parse::<i32>()
                .map_err(|_| ScheduleError::TaskFailed {
                    task: head.to_string(),
                    reason: "invalid priority".to_string(),
                })?;
            (name.trim().to_string(), priority)
        }
        None => (head.to_string(), 0),
    };

    let mut task = Task::new(name).with_priority(priority);
    for dep in deps {
        if !dep.is_empty() {
            task = task.depends_on(dep);
        }
    }
    Ok(task)
}

#[cfg(test)]
mod tests {
    use super::*;

    fn sample() -> Scheduler {
        let mut scheduler = Scheduler::new();
        scheduler.add(Task::new("fetch").with_priority(1)).unwrap();
        scheduler
            .add(Task::new("build").with_priority(5).depends_on("fetch"))
            .unwrap();
        scheduler
            .add(Task::new("test").depends_on("build"))
            .unwrap();
        scheduler
    }

    #[test]
    fn orders_dependencies_first() {
        let scheduler = sample();
        let order = scheduler.topological_order().unwrap();
        let fetch = order.iter().position(|n| n == "fetch").unwrap();
        let build = order.iter().position(|n| n == "build").unwrap();
        assert!(fetch < build);
    }

    #[test]
    fn rejects_duplicates() {
        let mut scheduler = Scheduler::new();
        scheduler.add(Task::new("compile")).unwrap();
        assert_eq!(
            scheduler.add(Task::new("compile")),
            Err(ScheduleError::DuplicateTask("compile".to_string()))
        );
    }

    #[test]
    fn detects_cycles() {
        let mut scheduler = Scheduler::new();
        scheduler.add(Task::new("a").depends_on("b")).unwrap();
        scheduler.add(Task::new("b").depends_on("a")).unwrap();
        assert!(matches!(
            scheduler.validate(),
            Err(ScheduleError::CycleDetected(_))
        ));
    }

    #[test]
    fn parses_expressions() {
        let task = parse_task("build(10): fetch, lint").unwrap();
        assert_eq!(task.name, "build");
        assert_eq!(task.priority, 10);
        assert_eq!(task.dependencies, vec!["fetch", "lint"]);
    }
}
